function updateClock() {


    var now = new Date();


    var h = now.getHours();

    var m = now.getMinutes();


    document.getElementById("clock").innerHTML =

        (h < 10 ? "0" : "") + h + ":" +

        (m < 10 ? "0" : "") + m;


    document.getElementById("date").innerHTML =

        days[now.getDay()] + ", " +

        now.getDate() + ". " +

        months[now.getMonth()] + " " +

        now.getFullYear();


}


function drawCalendar() {


    var now = new Date();


    var year = now.getFullYear();

    var month = now.getMonth();

    var today = now.getDate();


    document.getElementById("monthTitle").innerHTML =

        monthNames[month] + " " + year;


    var firstDay =

        new Date(year, month, 1).getDay();


    var daysInMonth =

        new Date(year, month + 1, 0).getDate();


    var offset = (firstDay + 6) % 7;


    var html = "";

    var day = 1;


    for (var row = 0; row < 6; row++) {


        html += "<tr>";


        for (var col = 0; col < 7; col++) {


            if ((row === 0 && col < offset) ||

                day > daysInMonth) {


                html += '<td class="empty"></td>';


            } else {


                var cls =

                    (day === today) ? "today" : "";


                html += '<td class="' +

                    cls + '">' + day + '</td>';


                day++;

            }

        }


        html += "</tr>";


        if (day > daysInMonth) {

            break;

        }

    }


    document.getElementById("calendarBody").innerHTML =

        html;

}


function weatherDescription(code) {


    if (code === 0) return "Vedro";

    if (code <= 3) return "Djelomično oblačno";

    if (code <= 48) return "Magla";

    if (code <= 67) return "Kiša";

    if (code <= 77) return "Snijeg";

    if (code <= 82) return "Pljuskovi";

    if (code <= 86) return "Snježni pljuskovi";

    if (code >= 95) return "Grmljavina";


    return "Promjenjivo vrijeme";

}


function loadWeather() {


    var url =

        "https://api.open-meteo.com/v1/forecast" +

        "?latitude=45.8150" +

        "&longitude=15.9819" +

        "&current=temperature_2m,weather_code" +

        "&daily=sunrise,sunset" +

        "&timezone=Europe%2FZagreb";


    var xhr = new XMLHttpRequest();


    xhr.open("GET", url, true);


    xhr.onreadystatechange = function() {


        if (xhr.readyState === 4) {


            if (xhr.status === 200) {


                try {


                    var data =

                        JSON.parse(xhr.responseText);


                    document.getElementById(

                        "temperature"

                    ).innerHTML =

                        Math.round(

                            data.current.temperature_2m

                        ) + "°";


                    document.getElementById(

                        "description"

                    ).innerHTML =

                        weatherDescription(

                            data.current.weather_code

                        );


                    document.getElementById(

                        "sunrise"

                    ).innerHTML =

                        data.daily.sunrise[0].substring(11, 16);


                    document.getElementById(

                        "sunset"

                    ).innerHTML =

                        data.daily.sunset[0].substring(11, 16);


                } catch(e) {


                    document.getElementById(

                        "description"

                    ).innerHTML =

                        "Weather unavailable";


                }


            } else {


                document.getElementById(

                    "description"

                ).innerHTML =

                    "Weather unavailable";


            }

        }

    };


    xhr.send();

}
